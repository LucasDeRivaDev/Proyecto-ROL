      document.getElementById("combatName").value = "";

      document.getElementById("combatIsPJ").checked = false;

    function startCombat() {
      combatState.combatants.sort((a, b) => b.des - a.des);
      combatState.currentIndex = 0;
      combatState.round = 1;
      combatState.started = true;
      renderCombatList();
      renderTurnIndicator();
    }

    function nextTurn() {
      if (!combatState.started || combatState.combatants.length === 0) return;
      combatState.combatants[combatState.currentIndex].guardOpen = false;
      let next = (combatState.currentIndex + 1) % combatState.combatants.length;
      let loops = 0;
      while (combatState.combatants[next].fallen && loops < combatState.combatants.length) {
        next = (next + 1) % combatState.combatants.length;
        loops++;
      }
      if (next <= combatState.currentIndex) combatState.round++;
      combatState.currentIndex = next;
      renderCombatList();
      renderTurnIndicator();
    }

    function resetCombat() {
      combatState = { combatants: [], currentIndex: -1, round: 0, started: false };
      renderCombatList();
      document.getElementById("combatTurnIndicator").innerHTML = "";
    }

    function applyDamage(index, amount) {
      const c = combatState.combatants[index];
      if (!c || c.fallen) return;
      c.pvCurrent = Math.max(0, c.pvCurrent - amount);
      if (c.pvCurrent <= 0) c.fallen = true;
      renderCombatList();
    }

    function applyHeal(index, amount) {
      const c = combatState.combatants[index];
      if (!c) return;
      c.pvCurrent = Math.min(c.pvMax, c.pvCurrent + amount);
      if (c.pvCurrent > 0) c.fallen = false;
      renderCombatList();
    }

    function applyBandage(index) {
      applyHeal(index, roll(4));
    }

    function applyCustomDmg(index) {
      const input = document.getElementById("dmgInput-" + index);
      const val = parseInt(input?.value) || 0;
      if (val > 0) applyDamage(index, val);
      else if (val < 0) applyHeal(index, Math.abs(val));
      if (input) input.value = "";
    }

    function removeCombatant(index) {
      combatState.combatants.splice(index, 1);
      if (combatState.currentIndex >= combatState.combatants.length) combatState.currentIndex = 0;
      renderCombatList();
    }

    }

    function rollFumble() {
      const idx = combatState.currentIndex;
      if (idx >= 0 && combatState.combatants[idx]) {
        combatState.combatants[idx].guardOpen = true;
        renderCombatList();
      }
      setBonusResult("criticalResult", "💀 Pifia (1 natural)\nEl combatiente quedó con la guardia abierta. Todos los ataques contra él tienen ventaja hasta su próximo turno.");
    }

    function rollQuickInstinct(index) {
      const c = combatState.combatants[index];
      if (!c) return;
      const d20 = roll(20);
      const total = d20;
      addDiceHistory(`Instintos rápido (${c.name}): d20 = ${d20} vs 11+: ${total >= 11 ? "ÉXITO" : "FALLO"}`);
      alert(`Instintos de ${c.name}: d20 = ${d20} vs 11+ → ${total >= 11 ? "✓ ÉXITO" : "✗ FALLO"}`);
    }

    function renderCombatList() {
      const el = document.getElementById("combatantList");
      if (!el) return;
      if (combatState.combatants.length === 0) {
        el.innerHTML = `<p style="font-size:0.85rem;color:rgba(241,230,200,0.6);text-align:center;padding:16px">Agregá combatientes para comenzar.</p>`;
        return;
      }
      el.innerHTML = combatState.combatants.map((c, i) => {
        const pct = Math.max(0, (c.pvCurrent / c.pvMax) * 100);
        const isActive = combatState.started && i === combatState.currentIndex;
        const barClass = pct <= 33 ? "hp-bar low" : "hp-bar";
        return `
          <div class="combatant-row ${isActive ? "active-turn" : ""} ${c.fallen ? "fallen" : ""}">
            <div>
              <div class="combatant-name">${isActive ? "▶ " : ""}${c.name} ${c.isPJ ? "🛡" : "👹"} ${c.guardOpen ? "⚠️ guardia abierta" : ""} ${c.fallen ? "💀 CAÍDO" : ""}</div>
              <div style="display:flex;align-items:center;gap:8px;margin-top:4px">
                <div class="hp-bar-wrap"><div class="${barClass}" style="width:${pct}%"></div></div>
                <span class="combatant-hp">${c.pvCurrent}/${c.pvMax} PV &nbsp; DEF ${c.def} &nbsp; DES ${c.des}</span>
              </div>
            </div>
            <div style="display:flex;gap:4px;flex-wrap:wrap;align-items:center">

    function renderTurnIndicator() {
